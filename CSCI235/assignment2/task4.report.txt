create or replace function applicant_skills (anumber decimal) returns text
language plpgsql as $$
<<body>>
declare
	n decimal := anumber;
begin
	-- for skill in skills_cur loop
	-- 	null;
	-- end loop;
	return array_to_string(array(
		select sname from spossessed t where t.anumber = n order by sname asc
	), ', ');
end
$$;
